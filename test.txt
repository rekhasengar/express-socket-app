    <script>
      // Retrieve the token from localStorage
      const token = localStorage.getItem('authToken');

      if (!token) {
        alert('You are not authenticated. Redirecting to login page.');
        window.location.href = '/login.html';
      } else {
        try {
          // Decode the token to extract user details
          const decodedToken = jwt_decode(token);
          const userEmail = decodedToken.email; // Assumes token has an 'email' field
          const emailDisplay = document.getElementById('userEmail');

          if (userEmail) {
            emailDisplay.textContent = `Logged in as: ${userEmail}`;
          } else {
            emailDisplay.textContent = 'Email not found in token.';
          }
        } catch (error) {
          console.error('Error decoding token:', error);
          alert('Invalid token. Redirecting to login page.');
          window.location.href = '/login.html';
        }

        // Connect to the backend using the token
        const socket = io('http://localhost:3000', {
          auth: {
            token: token,
          },
        });

        socket.on('connected', () => {
          console.log('Connected successfully!');
        });

        // Handle user going offline (tab or window close)
        window.addEventListener('beforeunload', () => {
          if (socket && socket.connected) {
            socket.emit('inactiveUser', { socketId: socket.id });
            console.log('User inactive event sent to the server:', { socketId: socket.id });
          }
        });

        // Listen for a message from the server
        socket.on('receiveMessage', (data) => {
          console.log('Received message from backend:', data.message.message);
        });

        socket.on('userStatus', (data) => {
          console.log('Received message from backend:', data.message.message);
        });

        socket.on('addUserInGroup', (data) => {
          console.log('Received message from backend:', data.message.message);
        });

        function displayMessage(messageText, type) {
          const messageContainer = document.getElementById('messageContainer');
          const messageElement = document.createElement('p');

          // Style the message based on the type
          switch (type) {
            case 'received':
              messageElement.style.color = 'blue';
              messageElement.textContent = `Received: ${messageText}`;
              break;
            case 'status':
              messageElement.style.color = 'green';
              messageElement.textContent = `Status Update: ${messageText}`;
              break;
            case 'group':
              messageElement.style.color = 'purple';
              messageElement.textContent = `Group Update: ${messageText}`;
              break;
            default:
              messageElement.textContent = messageText;
          }

          messageContainer.appendChild(messageElement);
        }

        // Send a message to the server
        document.getElementById('sendMessageButton').addEventListener('click', () => {
          const messageInput = document.getElementById('messageInput');
          const messageText = messageInput.value.trim();
          if (messageText === '') {
            alert('Message cannot be empty!');
            return;
          }

          const uuid = crypto.randomUUID();
          const decodedToken = jwt_decode(token);
          const userId = decodedToken.id;
          const payload = {
            eventType: 'sendMessage',
            socketId: socket.id, // Use the current socket ID
            data: {
              conversationId: '7e989195-3742-4f39-adf5-56b5508b4a18',
              senderId: userId,
              message: {
                id: uuid,
                message: messageText,
                timestamp: new Date().getTime(),
              },
            },
          };

          socket.emit('event', payload);
          console.log('Message sent successfully!');
          messageInput.value = ''; // Clear the input after sending the message
        });
      }
    </script>
